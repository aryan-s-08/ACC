1. Leaders in an Array
Approach: Right-to-Left Traversal • Time: O(N) • Space: O(N)
import java.util.*;

public class LeadersInArray {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        ArrayList<Integer> leaders = new ArrayList<>();
        int maxRight = arr[n - 1];
        leaders.add(maxRight);

        for (int i = n - 2; i >= 0; i--) {
            if (arr[i] >= maxRight) {
                leaders.add(arr[i]);
                maxRight = arr[i];
            }
        }

        Collections.reverse(leaders);
        for (int x : leaders) System.out.print(x + " ");
        sc.close();
    }
}

2. Block Swap Algorithm (Left Array Rotation)
Approach: In-place Iterative Block Swapping • Time: O(N) • Space: O(1)
import java.util.Scanner;

public class BlockSwapRotation {
    static void swap(int[] arr, int fi, int si, int d) {
        for (int i = 0; i < d; i++) {
            int temp = arr[fi + i];
            arr[fi + i] = arr[si + i];
            arr[si + i] = temp;
        }
    }

    static void leftRotate(int[] arr, int d, int n) {
        if (d == 0 || d == n) return;
        int i = d, j = n - d;
        while (i != j) {
            if (i < j) {
                swap(arr, d - i, d + j - i, i);
                j -= i;
            } else {
                swap(arr, d - i, d, j);
                i -= j;
            }
        }
        swap(arr, d - i, d, i);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
        int d = sc.nextInt() % n;

        leftRotate(arr, d, n);
        for (int x : arr) System.out.print(x + " ");
        sc.close();
    }
}

4. Majority Element (Boyer-Moore Voting)
Approach: Boyer-Moore Voting Algorithm • Time: O(N) • Space: O(1)
import java.util.Scanner;

public class MajorityElement {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        int candidate = 0, count = 0;
        for (int num : arr) {
            if (count == 0) candidate = num;
            count += (num == candidate) ? 1 : -1;
        }

        count = 0;
        for (int num : arr) {
            if (num == candidate) count++;
        }

        System.out.println((count > n / 2) ? candidate : -1);
        sc.close();
    }
}

5. Maximum Product Subarray
Approach: Min/Max Dynamic Programming • Time: O(N) • Space: O(1)
import java.util.Scanner;

public class MaxProductSubarray {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        int maxProd = arr[0], minProd = arr[0], result = arr[0];
        for (int i = 1; i < n; i++) {
            if (arr[i] < 0) {
                int temp = maxProd;
                maxProd = minProd;
                minProd = temp;
            }
            maxProd = Math.max(arr[i], maxProd * arr[i]);
            minProd = Math.min(arr[i], minProd * arr[i]);
            result = Math.max(result, maxProd);
        }

        System.out.println(result);
        sc.close();
    }
}

6. Maximum Sum of Hourglass in Matrix
Approach: Fixed-Pattern Matrix Traversal • Time: O(R × C) • Space: O(1)
import java.util.Scanner;

public class HourglassMaxSum {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt(), c = sc.nextInt();
        int[][] mat = new int[r][c];

        for (int i = 0; i < r; i++)
            for (int j = 0; j < c; j++)
                mat[i][j] = sc.nextInt();

        if (r < 3 || c < 3) {
            System.out.println(-1);
            sc.close();
            return;
        }

        int maxSum = Integer.MIN_VALUE;
        for (int i = 0; i <= r - 3; i++) {
            for (int j = 0; j <= c - 3; j++) {
                int sum = mat[i][j] + mat[i][j + 1] + mat[i][j + 2]
                        + mat[i + 1][j + 1]
                        + mat[i + 2][j] + mat[i + 2][j + 1] + mat[i + 2][j + 2];
                maxSum = Math.max(maxSum, sum);
            }
        }

        System.out.println(maxSum);
        sc.close();
    }
}

7. Longest Sequence of 1s After Flip
Approach: Two-Pointer Sliding Window • Time: O(N) • Space: O(1)
import java.util.Scanner;

public class LongestOnesAfterFlip {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        int left = 0, zeros = 0, maxLen = 0;
        for (int right = 0; right < n; right++) {
            if (arr[right] == 0) zeros++;
            while (zeros > 1) {
                if (arr[left] == 0) zeros--;
                left++;
            }
            maxLen = Math.max(maxLen, right - left + 1);
        }

        System.out.println(maxLen);
        sc.close();
    }
}

8. Maximum Equilibrium Sum / Balanced Partition
Approach: Running Prefix & Suffix Comparison • Time: O(N) • Space: O(N)
import java.util.Scanner;

public class MaxEquilibriumSum {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        long total = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextLong();
            total += arr[i];
        }

        long left = 0, ans = -1;
        for (int i = 0; i < n; i++) {
            left += arr[i];
            long right = total - left;
            if (left == right) {
                ans = Math.max(ans, left);
            }
        }

        System.out.println(ans);
        sc.close();
    }
}

9. Lexicographically First Palindromic String
Approach: Character Frequency Array Construction • Time: O(N) • Space: O(1)
import java.util.Scanner;

public class LexFirstPalindrome {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int[] freq = new int[26];
        for (char c : s.toCharArray()) freq[c - 'a']++;

        int oddCount = 0;
        char mid = 0;
        for (int i = 0; i < 26; i++) {
            if (freq[i] % 2 != 0) {
                oddCount++;
                mid = (char) (i + 'a');
            }
        }

        if (oddCount > 1) {
            System.out.println("Not Possible");
            sc.close();
            return;
        }

        StringBuilder left = new StringBuilder();
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < freq[i] / 2; j++) {
                left.append((char) (i + 'a'));
            }
        }

        StringBuilder result = new StringBuilder(left);
        if (oddCount == 1) result.append(mid);
        result.append(new StringBuilder(left).reverse());

        System.out.println(result.toString());
        sc.close();
    }
}

10. Karatsuba Multiplication Algorithm
Approach: Divide & Conquer BigInteger • Time: Θ(N1.585) • Space: O(log N)
import java.math.BigInteger;
import java.util.Scanner;

public class KaratsubaMultiplication {
    static BigInteger multiply(BigInteger x, BigInteger y) {
        int n = Math.max(x.toString().length(), y.toString().length());
        if (n <= 4) return x.multiply(y);

        int m = n / 2;
        BigInteger p = BigInteger.TEN.pow(m);

        BigInteger a = x.divide(p);
        BigInteger b = x.mod(p);
        BigInteger c = y.divide(p);
        BigInteger d = y.mod(p);

        BigInteger ac = multiply(a, c);
        BigInteger bd = multiply(b, d);
        BigInteger adbc = multiply(a.add(b), c.add(d)).subtract(ac).subtract(bd);

        return ac.multiply(BigInteger.TEN.pow(2 * m)).add(adbc.multiply(p)).add(bd);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        BigInteger x = sc.nextBigInteger();
        BigInteger y = sc.nextBigInteger();
        System.out.println(multiply(x, y));
        sc.close();
    }
}

11. Booth's Multiplication Algorithm
Approach: Bitwise Shift-and-Add Multiplication • Time: O(log B) • Space: O(1)
import java.util.Scanner;

public class BoothAlgo {
    public static int booth(int a, int b) {
        int sign = ((a < 0) ^ (b < 0)) ? -1 : 1;
        a = Math.abs(a);
        b = Math.abs(b);
        int res = 0;
        while (b != 0) {
            if ((b & 1) == 1) {
                res += a;
            }
            a <<= 1;
            b >>= 1;
        }
        return res * sign;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        System.out.println(booth(a, b));
        sc.close();
    }
}

12. Euclid's Algorithm (Greatest Common Divisor)
Approach: Euclidean Recursive Modulo • Time: O(log(min(A, B))) • Space: O(log(min(A, B)))
import java.util.Scanner;

public class EuclidAlgo {
    public static int gcd(int a, int b) {
        if (a == 0) return b;
        return gcd(b % a, a);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        System.out.println(gcd(a, b));
        sc.close();
    }
}
